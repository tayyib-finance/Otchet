'Забронирован' : 'В наличии'; let bookInfo = it.status === 'booked' ? `
Бронь: ${it.bookName} (${it.bookPhone})
` : ''; let storePriceHtml = it.storePrice ? `
На витрине: ${it.storePrice.toLocaleString('ru-RU')} ₽
` : ''; categoriesStock[it.category] += `
${title}
${it.category === 'iPhone' ? (it.sim || '') : it.category}${imeiText}

${(it.buy || 0).toLocaleString('ru-RU')} ₽
${statusBadge}
${it.status === 'in' ? `` : ''}
`; } // --- ОЖИДАЮТ ПЕРЕВОД --- else if (it.status === 'sold' && it.paymentMethod === 'Перевод' && !it.paymentReceived) { let buyerInfo = `Покупатель: ${it.buyerName || 'Не указан'} ${it.buyerPhone ? `(${it.buyerPhone})` : ''}`; let progressHtml = ''; if (paidAmount > 0) { const percent = Math.min(100, Math.round((paidAmount / it.sell) * 100)); progressHtml = `
Оплачено: ${paidAmount.toLocaleString('ru-RU')} из ${it.sell.toLocaleString('ru-RU')} ₽
`; } pendingHtml += `
${title}
${imeiText}
Продажа: ${(it.sell || 0).toLocaleString('ru-RU')} ₽
+${profit.toLocaleString('ru-RU')} ₽
${progressHtml}
Продан Ждёт перевод
${buyerInfo}
`; } // --- ПОЛНОСТЬЮ ОПЛАЧЕННЫЕ ПРОДАННЫЕ --- else { let buyerInfo = `Покупатель: ${it.buyerName || 'Не указан'} ${it.buyerPhone ? `(${it.buyerPhone})` : ''}`; soldHtml += `
${title}
${imeiText}
Продажа: ${(it.sell || 0).toLocaleString('ru-RU')} ₽
${profit >= 0 ? '+' : ''}${profit.toLocaleString('ru-RU')} ₽
Продан
${buyerInfo}
`; } }); let stockHtml = ''; const catOrder = ['iPhone', 'PlayStation', 'Ноутбук', 'Телевизор']; catOrder.forEach(cat => { if (categoriesStock[cat]) { stockHtml += `
📱 ${cat}
${categoriesStock[cat]}`; delete categoriesStock[cat]; } }); for (let cat in categoriesStock) { stockHtml += `
📱 ${cat}
${categoriesStock[cat]}`; } document.getElementById('stockList').innerHTML = stockHtml || '
Склад пуст
'; document.getElementById('soldList').innerHTML = soldHtml || '
Нет проданных
'; if (pendingHtml) { document.getElementById('pendingList').innerHTML = pendingHtml; document.getElementById('pendingWrapper').style.display = 'block'; } else { document.getElementById('pendingWrapper').style.display = 'none'; } document.getElementById('sumBuy').textContent = sumBuy.toLocaleString('ru-RU') + ' ₽'; document.getElementById('sumCash').textContent = sumCash.toLocaleString('ru-RU') + ' ₽'; const netProfit = totalProfit - totalExpenses; const pEl = document.getElementById('sumProfit'); pEl.textContent = (netProfit >= 0 ? '+' : '') + netProfit.toLocaleString('ru-RU') + ' ₽'; pEl.className = 'dash-value ' + (netProfit >= 0 ? 'green-text' : 'red-text'); }// ЛОГИКА ЧАСТИЧНОЙ ОПЛАТЫ window.openPartialPayModal = function(key) { const d = deals[key]; const paid = d.partialPaid || 0; const total = d.sell || 0; const remaining = total - paid; document.getElementById('partialPayKey').value = key; document.getElementById('partialPayInfo').innerHTML = `Нужно получить: ${remaining.toLocaleString('ru-RU')} ₽ из ${total.toLocaleString('ru-RU')} ₽`; const input = document.querySelector('#partialPayForm input[name="payAmount"]'); input.value = remaining; document.getElementById('partialPayModal').classList.add('active'); } window.closePartialPayModal = function() { document.getElementById('partialPayModal').classList.remove('active'); } document.getElementById('partialPayForm').addEventListener('submit', function(e) { e.preventDefault(); const key = document.getElementById('partialPayKey').value; const inputAmount = Number(document.querySelector('#partialPayForm input[name="payAmount"]').value); const d = deals[key]; const currentPaid = d.partialPaid || 0; const total = d.sell || 0; const newPaid = currentPaid + inputAmount; if (newPaid >= total) { update(ref(db, 'deals/' + key), { paymentReceived: true, partialPaid: newPaid }); alert('Сумма полностью оплачена!
